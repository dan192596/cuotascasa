```python
"""Engine A: scratch implementation of docs/algorithm.md for card W0-02 (decimal, prec 34).

Lives OUTSIDE the repository. Stdlib only, Python >= 3.9. Every rule cites its [ALG.*] id.
Field names follow the frozen contracts of W0-03 (packages/domain) and W0-04 (packages/schema).
"""
from __future__ import annotations

import calendar
import datetime as dt
import decimal
from decimal import ROUND_HALF_EVEN, ROUND_HALF_UP, Context, Decimal

# [ALG.CONV] 34 significant digits, ROUND_HALF_EVEN for every operation that is not HALF_UP_2.
decimal.setcontext(Context(prec=34, rounding=ROUND_HALF_EVEN))
CENT = Decimal("0.01")
ZERO = Decimal("0")
ONE = Decimal("1")
TWELVE = Decimal("12")


class AlgError(Exception):
    """Typed error: type_name is the DomainError class, rule the [ALG.*] id that demands it."""

    def __init__(self, type_name: str, rule: str, k: int | None = None):
        super().__init__(f"{type_name} {rule} k={k}")
        self.type_name, self.rule, self.k = type_name, rule, k


def D(s: str) -> Decimal:
    """Strict decimal-string parse: rejects non-strings and exponent notation."""
    if not isinstance(s, str) or "e" in s.lower():
        raise TypeError(f"decimal string expected: {s!r}")
    return Decimal(s)


def half_up_2(x: Decimal) -> Decimal:
    """[ALG.CONV] HALF_UP_2: 2 decimals, half away from zero."""
    return x.quantize(CENT, rounding=ROUND_HALF_UP)


def money(x: Decimal) -> str:
    if x != x.quantize(CENT):
        raise ValueError(f"not a cent amount: {x}")
    return f"{x.quantize(CENT):f}"


# ------------------------------------------------------------------ [ALG.DATES]
def parse_date(s: str) -> dt.date:
    return dt.date.fromisoformat(s)


def due_day(y: int, m: int, payment_day) -> int:
    dim = calendar.monthrange(y, m)[1]
    return dim if payment_day == "END_OF_MONTH" else min(int(payment_day), dim)


def check_first_due(first_due: dt.date, payment_day) -> None:
    if first_due.day != due_day(first_due.year, first_due.month, payment_day):
        raise AlgError("InvalidInputError", "ALG.DATES")


def due_date(first_due: dt.date, payment_day, k: int) -> dt.date:
    y, m0 = divmod(first_due.year * 12 + first_due.month - 1 + (k - 1), 12)
    return dt.date(y, m0 + 1, due_day(y, m0 + 1, payment_day))


def k_for_date(first_due: dt.date, payment_day, d: dt.date) -> int:
    """[ALG.EVENTS.ANCHOR] rule 2: first installment whose due date is >= d."""
    k = 1
    while due_date(first_due, payment_day, k) < d:
        k += 1
    return k


# ------------------------------------------------------------------ [ALG.LEVEL] [ALG.ZERO]
def periodic_rate(i: Decimal, fs: list) -> Decimal:
    """[ALG.TERMS] r = (i + f) / 12, never rounded."""
    return (i + sum(fs, ZERO)) / TWELVE


def level_payment(B: Decimal, r: Decimal, m: int) -> Decimal:
    if m < 1:
        raise AlgError("InvalidInputError", "ALG.LEVEL")
    if r == ZERO:
        return half_up_2(B / Decimal(m))
    P = (ONE + r) ** m
    return half_up_2((B * r) / (ONE - (ONE / P)))


# ------------------------------------------------------------------ one installment
def split_insurance(ins: Decimal, fs: list) -> list:
    """[ALG.PERIOD.SPLIT]; [ALG.ZERO]: no split when f = 0."""
    if not fs:
        return []
    f = sum(fs, ZERO)
    if f == ZERO:
        return [ZERO for _ in fs]
    parts = [half_up_2((ins * fj) / f) for fj in fs[:-1]]
    return parts + [ins - sum(parts, ZERO)]


def normal_period(B, i, fs, level, profile):
    """(interest, components, capital, financialCharge) of an installment computed as normal."""
    f = sum(fs, ZERO)
    r = (i + f) / TWELVE
    if r == ZERO:  # [ALG.ZERO]
        return ZERO, [ZERO for _ in fs], level, ZERO
    if profile == "FHA_GT_V1":  # [ALG.PERIOD.FHA_GT_V1]
        charge = half_up_2(B * r)
        interest = charge if f == ZERO else half_up_2((charge * i) / (i + f))
        return interest, split_insurance(charge - interest, fs), level - charge, charge
    if profile == "SIMPLE":  # [ALG.PERIOD.SIMPLE]
        interest = half_up_2((B * i) / TWELVE)
        parts = [half_up_2((B * fj) / TWELVE) for fj in fs]
        fc = interest + sum(parts, ZERO)
        return interest, parts, level - fc, fc
    raise AlgError("InvalidInputError", "ALG.TERMS")


def last_period(B, i, fs, profile):
    """[ALG.LAST]: (interest, components, capital = B)."""
    interest = half_up_2((B * i) / TWELVE)
    if profile == "FHA_GT_V1":
        parts = split_insurance(half_up_2((B * sum(fs, ZERO)) / TWELVE), fs)
    else:
        parts = [half_up_2((B * fj) / TWELVE) for fj in fs]
    return interest, parts, B


# ------------------------------------------------------------------ [ALG.TERM] state
class State:
    """PeriodState after paying installment k (phase 3 included), plus the fixed charges."""

    def __init__(self, balance, i, fs, level, profile, k, mode, term, fixed):
        self.balance, self.i, self.fs, self.level = balance, i, list(fs), level
        self.profile, self.k, self.mode, self.term, self.fixed = profile, k, mode, term, list(fixed)


def is_last_row(st: State, k: int, B: Decimal, fc: Decimal) -> bool:
    """[ALG.LAST.FIXED_TERM] and [ALG.LAST.DERIVED_TERM], both with financialCharge."""
    return st.level - fc >= B or (st.mode == "fixed" and k >= st.term)


def simulate_capitals(st: State, limit=None) -> list:
    """Capital of installments k+1 .. last of the current calendar, without future events."""
    B, k, out = st.balance, st.k, []
    while B > ZERO and (limit is None or len(out) < limit):
        k += 1
        _, _, capital, fc = normal_period(B, st.i, st.fs, st.level, st.profile)
        if is_last_row(st, k, B, fc):
            out.append(B)
            break
        if capital <= ZERO and st.mode == "derived":  # a derived term would never end
            raise AlgError("NegativeAmortizationError", "ALG.TERM", k)
        out.append(capital)
        B -= capital
    return out


def remaining_term(st: State) -> int:
    """[ALG.TERM] remainingTerm(state)."""
    return len(simulate_capitals(st))


def project_capital(st: State, n: int) -> Decimal:
    """[ALG.TERM] projectCapital(state, n)."""
    return sum(simulate_capitals(st, n), ZERO)


def current_term(st: State) -> int:
    """[ALG.TERM] fixed: the datum; derived: k + remainingTerm(state)."""
    return st.term if st.mode == "fixed" else st.k + remaining_term(st)


# ------------------------------------------------------------------ [ALG.EVENTS]
PHASE = {"ReportedBalance": (0, 0), "RateChange": (1, 0), "FixedChargeChange": (1, 1),
         "Prepayment": (3, 0), "AdvanceInstallments": (3, 1), "ActualPayment": (4, 0)}
COMPONENTS = ("capital", "interest", "insurance", "fixedCharges")


def assign_k(terms: dict, ev: dict) -> int:
    """[ALG.EVENTS.ANCHOR]: installmentNumber wins; otherwise first due date >= event date."""
    if ev.get("installmentNumber") is not None:
        return int(ev["installmentNumber"])
    if ev["type"] == "ActualPayment":
        raise AlgError("InvalidInputError", "ALG.EVENTS.ANCHOR")
    return k_for_date(parse_date(terms["firstDueDate"]), terms["paymentDay"], parse_date(ev["date"]))


def order_key(terms: dict, ev: dict):
    """[ALG.EVENTS.ORDER] (k, phase, date, typeRank, id)."""
    phase, rank = PHASE[ev["type"]]
    return (assign_k(terms, ev), phase, ev["date"], rank, ev["id"])


def build_schedule(terms: dict, events=None) -> dict:
    events = list(events or [])
    first_due, pday = parse_date(terms["firstDueDate"]), terms["paymentDay"]
    check_first_due(first_due, pday)
    i, fs = D(terms["interestRate"]), [D(x) for x in terms["insuranceRates"]]
    P0, n0 = D(terms["principal"]), int(terms["termMonths"])
    fixed = [(c["label"], D(c["amount"]), parse_date(c["effectiveFrom"])) for c in terms["fixedCharges"]]
    level0 = level_payment(P0, periodic_rate(i, fs), n0)
    st = State(P0, i, fs, level0, terms["roundingProfile"], 0, "fixed", n0, fixed)
    by_k: dict = {}
    for e in sorted(events, key=lambda x: order_key(terms, x)):
        k_e = assign_k(terms, e)
        if e.get("installmentNumber") is not None and k_e < 1:
            raise AlgError("InvalidInputError", "ALG.EVENTS.ANCHOR", k_e)
        by_k.setdefault(k_e, []).append(e)
    rows, anchors, deltas, k = [], [], [], 0
    while st.balance > ZERO:
        k += 1
        due = due_date(first_due, pday, k)
        evs = by_k.get(k, [])
        if any(e.get("installmentNumber") is not None for e in evs) and k > current_term(st):
            raise AlgError("InvalidInputError", "ALG.EVENTS.ANCHOR", k)
        rb = [e for e in evs if e["type"] == "ReportedBalance"]  # phase 0 [ALG.ANCHOR]
        if rb:
            for e in rb:  # AnchorRealDelta (W0-03)
                anchors.append({"eventId": e["id"], "k": k, "reported": e["balance"],
                                "projected": money(st.balance),
                                "realDelta": money(D(e["balance"]) - st.balance)})
            st.balance = D(max(rb, key=lambda e: (e["date"], e["id"]))["balance"])
        for e in evs:  # phase 1
            if e["type"] == "RateChange":  # [ALG.RATE]
                term_now = current_term(st)
                if e.get("interestRate") is not None:
                    st.i = D(e["interestRate"])
                if e.get("insuranceRates") is not None:
                    st.fs = [D(x) for x in e["insuranceRates"]]
                r2 = periodic_rate(st.i, st.fs)
                if e["policy"] == "RECALC_INSTALLMENT_KEEP_TERM":
                    st.level = level_payment(st.balance, r2, term_now - (k - 1))
                    st.mode, st.term = "fixed", term_now
                else:
                    rule = {"KEEP_INSTALLMENT_ADJUST_TERM": "ALG.RATE.KEEP_INSTALLMENT",
                            "BANK_INSTALLMENT": "ALG.RATE.BANK_INSTALLMENT"}[e["policy"]]
                    if e["policy"] == "BANK_INSTALLMENT":
                        st.level = D(e["bankInstallment"])
                    if st.level - half_up_2(st.balance * r2) <= ZERO:
                        raise AlgError("NegativeAmortizationError", rule, k)
                    st.mode = "derived"
            elif e["type"] == "FixedChargeChange":  # [ALG.FIXEDCHANGE]
                st.fixed = [(c["label"], D(c["amount"]), due) for c in e["fixedCharges"]]
        B = st.balance  # phase 2
        interest, parts, capital, fc = normal_period(B, st.i, st.fs, st.level, st.profile)
        last = is_last_row(st, k, B, fc)
        if last:
            interest, parts, capital = last_period(B, st.i, st.fs, st.profile)
        elif capital <= ZERO and st.mode == "derived":  # a derived term would never end
            raise AlgError("NegativeAmortizationError", "ALG.TERM", k)
        fixed_k = sum((a for _, a, eff in st.fixed if eff <= due), ZERO)  # [ALG.FIXED]
        insurance = sum(parts, ZERO)
        row = {"k": k, "dueDate": due.isoformat(), "opening": B, "level": st.level,
               "interest": interest, "insurance": insurance, "insuranceComponents": parts,
               "capital": capital, "fixedCharges": fixed_k,
               "total": capital + interest + insurance + fixed_k, "closing": B - capital,
               "prepayment": ZERO, "commission": ZERO, "paid": False}
        st.balance, st.k = B - capital, k
        for e in evs:  # phase 3 [ALG.PREPAY] [ALG.ADVANCE]
            if e["type"] not in ("Prepayment", "AdvanceInstallments") or st.balance == ZERO:
                continue
            if e["type"] == "Prepayment":
                amount, mode = D(e["amount"]), e["mode"]
            else:
                amount, mode = project_capital(st, int(e["count"])), "REDUCE_TERM"
            applied = min(amount, st.balance)  # [ALG.PREPAY.CAP]
            com = e.get("commission")  # [ALG.PREPAY.COMMISSION]; absent = no commission
            fee = ZERO if com is None else (
                D(com["amount"]) if com["kind"] == "FLAT" else half_up_2(applied * D(com["rate"])))
            term_before = current_term(st)
            st.balance -= applied
            row["prepayment"] += applied
            row["commission"] += fee
            if st.balance == ZERO:
                break
            if mode == "REDUCE_TERM":
                st.mode = "derived"
            else:
                st.level = level_payment(st.balance, periodic_rate(st.i, st.fs), term_before - k)
                st.mode, st.term = "fixed", term_before
        for e in evs:  # phase 4 [ALG.ACTUAL]
            if e["type"] == "ActualPayment":
                row["paid"] = True
                if e.get("breakdown") is not None:  # ComponentRealDelta (W0-03): the four components
                    proj = {"capital": capital, "interest": interest, "insurance": insurance,
                            "fixedCharges": fixed_k}
                    delta = {"eventId": e["id"], "k": k}
                    delta.update({c: money(D(e["breakdown"][c]) - proj[c]) for c in COMPONENTS})
                    deltas.append(delta)
        rows.append(row)
        if last:
            break
    pending = [kk for kk, evs in by_k.items() if kk > k and any(e.get("installmentNumber") is not None for e in evs)]
    if pending:
        raise AlgError("InvalidInputError", "ALG.EVENTS.ANCHOR", min(pending))
    return {"level": level0, "rows": rows, "anchors": anchors, "componentDeltas": deltas}


# ------------------------------------------------------------------ summaries
def totals(rows: list) -> dict:
    """ScheduleTotals (W0-03); totalPaid per [ALG.METRICS]."""
    def s(key):
        return sum((r[key] for r in rows), ZERO)
    out = {"interest": s("interest"), "insurance": s("insurance"), "capital": s("capital"),
           "fixedCharges": s("fixedCharges"), "prepayments": s("prepayment"), "commissions": s("commission"),
           "total": s("total")}
    out["totalPaid"] = out["total"] + out["prepayments"] + out["commissions"]
    return out


YEARLY_KEYS = (("capital", "capital"), ("interest", "interest"), ("insurance", "insurance"),
               ("fixedCharges", "fixedCharges"), ("prepayments", "prepayment"),
               ("commissions", "commission"), ("total", "total"))


def yearly_subtotals(rows: list) -> list:
    """[ALG.YEARLY] by calendar year of dueDate."""
    years: dict = {}
    for r in rows:
        acc = years.setdefault(int(r["dueDate"][:4]), {k: ZERO for k, _ in YEARLY_KEYS})
        for key, src in YEARLY_KEYS:
            acc[key] += r[src]
    return [dict({"year": y}, **v) for y, v in sorted(years.items())]


def path_summary(rows: list) -> dict:
    t = totals(rows)
    return {"installments": len(rows), "endDate": rows[-1]["dueDate"],
            "interestAndInsurance": t["interest"] + t["insurance"], "totalPaid": t["totalPaid"]}


def compare(base_rows: list, scen_rows: list) -> dict:
    """[ALG.METRICS] scenario against its base."""
    b, s = path_summary(base_rows), path_summary(scen_rows)
    return {"interestSaved": b["interestAndInsurance"] - s["interestAndInsurance"],
            "monthsSaved": b["installments"] - s["installments"],
            "netSaving": b["totalPaid"] - s["totalPaid"], "base": b, "scenario": s}


# ------------------------------------------------------------------ [ALG.PATHS]
CUTOFF_TYPES = ("ReportedBalance", "ActualPayment", "Prepayment", "AdvanceInstallments")


def cutoff_k(terms: dict, real_events: list) -> int:
    """[ALG.PATHS.CUTOFF]."""
    return max([assign_k(terms, e) for e in real_events if e["type"] in CUTOFF_TYPES] or [0])


def build_paths(terms: dict, real_events: list, scenario_events) -> dict:
    """PathsInput (W0-03): scenario_events = None means there is no scenario."""
    ck = cutoff_k(terms, real_events)
    for e in scenario_events or []:
        if assign_k(terms, e) <= ck:
            raise AlgError("InvalidInputError", "ALG.PATHS.CUTOFF", assign_k(terms, e))
    scenario = None if scenario_events is None else build_schedule(terms, list(real_events) + list(scenario_events))
    return {"cutoffK": ck, "original": build_schedule(terms, []),
            "real": build_schedule(terms, real_events), "scenario": scenario}


# ------------------------------------------------------------------ [ALG.GOAL]
def goal_seek(terms: dict, real_events: list, scenario_events, request: dict) -> dict:
    """GoalSeekRequest (W0-03): {basePath, prepaymentDate, goal: {kind, date | amount}}."""
    date, goal = request["prepaymentDate"], request["goal"]
    k = k_for_date(parse_date(terms["firstDueDate"]), terms["paymentDay"], parse_date(date))
    if k <= cutoff_k(terms, real_events):
        raise AlgError("InfeasibleGoalError", "ALG.GOAL", k)
    base_events = list(real_events)
    if request["basePath"] == "SCENARIO":
        base_events += list(scenario_events or [])
    base = build_schedule(terms, base_events)["rows"]
    mode = "REDUCE_TERM" if goal["kind"] == "FINISH_BY" else "REDUCE_INSTALLMENT"
    closing_k = base[k - 1]["closing"]

    def scenario(cents: int) -> list:
        p = {"id": "goal-seek", "type": "Prepayment", "date": date,
             "amount": money(Decimal(cents) / 100), "mode": mode}
        return build_schedule(terms, base_events + [p])["rows"]

    def meets(rows: list) -> bool:
        if goal["kind"] == "FINISH_BY":
            return rows[-1]["dueDate"] <= goal["date"]
        return len(rows) <= k or rows[k]["total"] <= D(goal["amount"])

    if meets(base):
        return {"k": k, "closingK": closing_k, "kind": "ALREADY_MET"}
    hi = int(closing_k * 100)
    if not meets(scenario(hi)):
        return {"k": k, "closingK": closing_k, "kind": "INFEASIBLE", "payoffAmount": closing_k,
                "reason": "GOAL_DATE_BEFORE_PREPAYMENT"}
    lo = 1
    while lo < hi:
        mid = (lo + hi) // 2
        if meets(scenario(mid)):
            hi = mid
        else:
            lo = mid + 1
    amount = Decimal(lo) / 100
    return {"k": k, "closingK": closing_k, "kind": "FOUND", "amount": amount,
            "metrics": compare(base, scenario(lo)), "isPayoff": amount == closing_k}


# ------------------------------------------------------------------ [ALG.VALIDATE]
def validate(terms: dict, real_events: list, reported: dict) -> dict:
    """TemplateValidationRequest (W0-03): reported is a ReportedBalance event."""
    k = assign_k(terms, reported)
    kept = [e for e in real_events if not (e["type"] == "ReportedBalance" and assign_k(terms, e) >= k)]
    opening = {r["k"]: r["opening"] for r in build_schedule(terms, kept)["rows"]}
    Br = D(reported["balance"])
    delta = Br - opening[k]
    amber = max(Decimal("50.00"), Decimal("0.0002") * Br)
    if abs(delta) <= Decimal("1.00"):
        status, cause = "GREEN", None
    else:
        status = "AMBER" if abs(delta) <= amber else "RED"
        near = [opening[j] for j in (k - 1, k + 1) if j in opening]
        cause = "INSTALLMENT_MISALIGNMENT" if any(abs(Br - o) <= Decimal("1.00") for o in near) else "UNKNOWN"
    return {"k": k, "reported": Br, "modeled": opening[k], "realDelta": delta, "status": status, "cause": cause,
            "modeledPrevious": opening.get(k - 1), "modeledNext": opening.get(k + 1), "amberLimit": amber}
```

```python
def no_floats(text: str):
    def boom(token):
        raise AssertionError(f"JSON float {token}")
    return json.loads(text, parse_float=boom)


def headline(case: dict) -> tuple:
    e = case["expected"]
    if "error" in e:
        return ("error", e["error"]["type"], e["error"]["rule"], str(e["error"].get("k", "")))
    if case["operation"] == "buildSchedule":
        return (e["rows"][0]["level"], str(e["installmentCount"]), e["endDate"], e["totals"]["totalPaid"])
    if case["operation"] == "buildPaths":
        return (str(e["cutoffK"]), str(e["real"]["installmentCount"]), str(e["scenario"]["installmentCount"]),
                e["metrics"]["netSaving"])
    if case["operation"] == "goalSeek":
        return (e["kind"], e.get("amount") or e.get("payoffAmount") or case["context"]["closingK"])
    return (e["status"], str(e["cause"]), e["realDelta"], case["context"]["amberLimit"])


def all_keys(node) -> set:
    if isinstance(node, dict):
        return set(node) | {k for v in node.values() for k in all_keys(v)}
    if isinstance(node, list):
        return {k for v in node for k in all_keys(v)}
    return set()
```

```python
"""Writes every example JSON (engine A) into OUT/core and OUT/events.

Usage: python3 emit.py OUT
`expected` uses the names of the domain outputs (W0-03): Schedule, ScheduleRow, ScheduleTotals, RealDelta,
YearlySubtotal, ComparisonMetrics, GoalSeekResult and TemplateValidationResult. Values that no domain function
returns go to the case's `context`.
"""
from __future__ import annotations

import json
import sys
from decimal import Decimal
from pathlib import Path

import engine as E
from catalog import EXAMPLES

SCHEMA = "cuotascasa/algorithm-example@1"
IMPLICIT = {"goalSeek": ["Prepayment"], "validateAgainstReportedBalance": ["ReportedBalance"]}
EVENT_LISTS = ("events", "realEvents", "scenarioEvents")


def m(x):
    """Decimal -> 2-decimal string; lists recursively; None stays None."""
    if x is None:
        return None
    if isinstance(x, list):
        return [m(v) for v in x]
    return E.money(x)


def row_json(r: dict) -> dict:
    out = {}
    for key, val in r.items():
        out[key] = val if key in ("k", "dueDate", "paid") else m(val)
    return out


def totals_json(t: dict) -> dict:
    return {key: m(val) for key, val in t.items()}


def metrics_json(c: dict, currency: str) -> dict:
    """ComparisonMetrics (W0-03) of a scenario against its base."""
    return {"currency": currency, "interestSaved": m(c["interestSaved"]), "monthsSaved": c["monthsSaved"],
            "baseEndDate": c["base"]["endDate"], "endDate": c["scenario"]["endDate"],
            "baseTotalPaid": m(c["base"]["totalPaid"]), "totalPaid": m(c["scenario"]["totalPaid"]),
            "netSaving": m(c["netSaving"])}


def real_delta_json(res: dict) -> dict:
    """RealDelta (W0-03): AnchorRealDelta and ComponentRealDelta lists."""
    return {"perAnchor": res["anchors"], "perComponent": res["componentDeltas"]}


def pick(rows: list, show: list) -> list:
    n = len(rows)
    ks = [n if s == "last" else n - 1 if s == "last-1" else s for s in show]
    return [row_json(rows[k - 1]) for k in ks]


def schedule_json(res: dict, show: list, yearly: bool) -> dict:
    rows = res["rows"]
    out = {"installmentCount": len(rows), "endDate": rows[-1]["dueDate"], "rows": pick(rows, show),
           "totals": totals_json(E.totals(rows))}
    if yearly:
        out["yearly"] = [{key: (val if key == "year" else m(val)) for key, val in y.items()}
                         for y in E.yearly_subtotals(rows)]
    return out


def expected(case: dict) -> tuple:
    """Returns (expected, context); context is None when the operation has none."""
    op, t = case["operation"], case["terms"]
    try:
        if op == "buildSchedule":
            res = E.build_schedule(t, case["events"])
            out = schedule_json(res, case["show"], case.get("yearly", False))
            if res["anchors"] or res["componentDeltas"]:
                out["realDelta"] = real_delta_json(res)
            if case.get("compareToNoEvents"):
                out["comparedToNoEvents"] = metrics_json(
                    E.compare(E.build_schedule(t, [])["rows"], res["rows"]), t["currency"])
            return out, None
        if op == "buildPaths":
            p = E.build_paths(t, case["realEvents"], case["scenarioEvents"])
            original = p["original"]["rows"]
            return {"cutoffK": p["cutoffK"],
                    "original": {"installmentCount": len(original), "endDate": original[-1]["dueDate"],
                                 "totals": totals_json(E.totals(original))},
                    "real": schedule_json(p["real"], case["show"], False),
                    "scenario": schedule_json(p["scenario"], case["show"], case.get("yearly", False)),
                    "realDelta": real_delta_json(p["real"]),
                    "metrics": metrics_json(E.compare(p["real"]["rows"], p["scenario"]["rows"]), t["currency"])}, None
        if op == "goalSeek":
            g = E.goal_seek(t, case["realEvents"], case["scenarioEvents"], case["request"])
            out = {"kind": g["kind"]}
            if g["kind"] == "FOUND":
                out.update(amount=m(g["amount"]), metrics=metrics_json(g["metrics"], t["currency"]),
                           isPayoff=g["isPayoff"])
            if g["kind"] == "INFEASIBLE":
                out.update(payoffAmount=m(g["payoffAmount"]), reason=g["reason"])
            return out, {"k": g["k"], "closingK": m(g["closingK"])}
        if op == "validateAgainstReportedBalance":
            v = E.validate(t, case["realEvents"], case["reported"])
            return ({"k": v["k"], "reported": m(v["reported"]), "modeled": m(v["modeled"]),
                     "realDelta": m(v["realDelta"]), "status": v["status"], "cause": v["cause"]},
                    {"modeledPrevious": m(v["modeledPrevious"]), "modeledNext": m(v["modeledNext"]),
                     "amberLimit": f"{v['amberLimit'].quantize(Decimal('0.000001')):f}"})
    except E.AlgError as err:
        out = {"type": err.type_name, "rule": err.rule}
        if err.k is not None:
            out["k"] = err.k
        return {"error": out}, None
    raise ValueError(f"unknown operation {op}")


def event_types(ex: dict) -> list:
    types = set()
    for c in ex["cases"]:
        for key in EVENT_LISTS:
            types.update(e["type"] for e in (c.get(key) or []))
        types.update(IMPLICIT.get(c["operation"], []))
    return sorted(types)


def case_json(case: dict) -> dict:
    out = {key: val for key, val in case.items()
           if key not in ("show", "yearly", "compareToNoEvents")}
    out["expected"], context = expected(case)
    if context is not None:
        out["context"] = context
    return out


def example_json(ex: dict) -> dict:
    return {"synthetic": True, "schema": SCHEMA, "id": ex["id"], "title": ex["title"],
            "pendingItems": ex["pendingItems"], "sections": ex["sections"],
            "eventTypes": event_types(ex), "cases": [case_json(c) for c in ex["cases"]]}


def dump(obj) -> str:
    return json.dumps(obj, ensure_ascii=False, indent=2) + "\n"


def main(out_dir: str) -> None:
    out = Path(out_dir)
    for ex in EXAMPLES:
        doc = example_json(ex)
        folder = out / ex["set"]
        folder.mkdir(parents=True, exist_ok=True)
        (folder / f"{ex['id']}.json").write_text(dump(doc), encoding="utf-8")
        print(f"{ex['set']}/{ex['id']}.json eventTypes={doc['eventTypes']} cases={len(doc['cases'])}")


if __name__ == "__main__":
    main(sys.argv[1])
```

```python
"""Engine B: independent recomputation of every example JSON with exact rationals.

Written from docs/algorithm.md only; it imports nothing from engine.py, catalog.py or emit.py.
Usage: python3 recompute.py DIR   (DIR holds core/ and events/)
Prints one line per example: '<id>: recomputed: yes' or the first differences; exit 1 on any difference.
It recomputes both `expected` (domain names of W0-03) and `context` of every case.
"""
from __future__ import annotations

import calendar
import datetime as dt
import json
import sys
from fractions import Fraction as F
from pathlib import Path

HUNDRED = F(100)


def num(s: str) -> F:
    if not isinstance(s, str):
        raise TypeError(f"decimal string expected: {s!r}")
    return F(s)


def hu2(q: F) -> F:
    """HALF_UP_2 on an exact rational: half away from zero."""
    sign = -1 if q < 0 else 1
    n = abs(q) * HUNDRED
    whole = n.numerator // n.denominator
    if n - whole >= F(1, 2):
        whole += 1
    return F(sign * whole, 100)


def txt(q: F, places: int = 2) -> str:
    scale = 10 ** places
    n = q * scale
    if n.denominator != 1:
        raise ValueError(f"{q} has more than {places} decimals")
    v = n.numerator
    sign = "-" if v < 0 else ""
    v = abs(v)
    return f"{sign}{v // scale}.{v % scale:0{places}d}"


def due(first: str, pday, k: int) -> str:
    y, mth = int(first[:4]), int(first[5:7]) + k - 1
    y, mth = y + (mth - 1) // 12, (mth - 1) % 12 + 1
    last = calendar.monthrange(y, mth)[1]
    day = last if pday == "END_OF_MONTH" else min(int(pday), last)
    return dt.date(y, mth, day).isoformat()


def k_of(terms: dict, e: dict) -> int:
    if e.get("installmentNumber") is not None:
        return e["installmentNumber"]
    if e["type"] == "ActualPayment":
        raise Err("InvalidInputError", "ALG.EVENTS.ANCHOR", None)
    k = 1
    while due(terms["firstDueDate"], terms["paymentDay"], k) < e["date"]:
        k += 1
    return k


class Err(Exception):
    def __init__(self, typ, rule, k):
        super().__init__(typ)
        self.as_json = {"type": typ, "rule": rule} if k is None else {"type": typ, "rule": rule, "k": k}


def lvl(B: F, rate: F, m: int) -> F:
    if rate == 0:
        return hu2(B / m)
    return hu2(B * rate / (1 - (1 + rate) ** (-m)))


def split(ins: F, comps: list) -> list:
    tot = sum(comps, F(0))
    if not comps:
        return []
    if tot == 0:
        return [F(0)] * len(comps)
    head = [hu2(ins * c / tot) for c in comps[:-1]]
    return head + [ins - sum(head, F(0))]


def compute_row(B, i, comps, L, prof):
    """Returns (interest, parts, capital, financialCharge) of a normal installment."""
    f = sum(comps, F(0))
    if i + f == 0:
        return F(0), [F(0)] * len(comps), L, F(0)
    if prof == "FHA_GT_V1":
        ch = hu2(B * (i + f) / 12)
        it = ch if f == 0 else hu2(ch * i / (i + f))
        return it, split(ch - it, comps), L - ch, ch
    it = hu2(B * i / 12)
    ps = [hu2(B * c / 12) for c in comps]
    return it, ps, L - it - sum(ps, F(0)), it + sum(ps, F(0))


def final_row(B, i, comps, prof):
    it = hu2(B * i / 12)
    if prof == "FHA_GT_V1":
        return it, split(hu2(B * sum(comps, F(0)) / 12), comps)
    return it, [hu2(B * c / 12) for c in comps]


def count_rest(B, i, comps, L, prof, k, fixed_term):
    """Rows k+1 .. last; fixed_term None means derived. Returns list of capitals."""
    caps = []
    while B > 0:
        k += 1
        _, _, cap, fc = compute_row(B, i, comps, L, prof)
        if L - fc >= B or (fixed_term is not None and k >= fixed_term):
            caps.append(B)
            return caps
        if fixed_term is None and cap <= 0:
            raise Err("NegativeAmortizationError", "ALG.TERM", k)
        caps.append(cap)
        B -= cap
    return caps


ORDER = {"ReportedBalance": (0, 0), "RateChange": (1, 0), "FixedChargeChange": (1, 1),
         "Prepayment": (3, 0), "AdvanceInstallments": (3, 1), "ActualPayment": (4, 0)}


def run(terms: dict, events: list) -> dict:
    first, pday = terms["firstDueDate"], terms["paymentDay"]
    if first != due(first, pday, 1):
        raise Err("InvalidInputError", "ALG.DATES", None)
    i = num(terms["interestRate"])
    comps = [num(x) for x in terms["insuranceRates"]]
    B = num(terms["principal"])
    n = terms["termMonths"]
    prof = terms["roundingProfile"]
    charges = [(num(c["amount"]), c["effectiveFrom"]) for c in terms["fixedCharges"]]
    L = lvl(B, (i + sum(comps, F(0))) / 12, n)
    fixed_term = n  # None once the term is derived
    keyed = sorted(((k_of(terms, e),) + ORDER[e["type"]][:1] + (e["date"], ORDER[e["type"]][1], e["id"]), e)
                   for e in events)
    for (kk, *_), e in keyed:
        if e.get("installmentNumber") is not None and kk < 1:
            raise Err("InvalidInputError", "ALG.EVENTS.ANCHOR", kk)
    rows, anchors, deltas = [], [], []
    k = 0

    def term_now():
        return fixed_term if fixed_term is not None else (k - 1) + len(count_rest(B, i, comps, L, prof, k - 1, None))

    while B > 0:
        k += 1
        d = due(first, pday, k)
        here = [e for (kk, *_), e in keyed if kk == k]
        if any(e.get("installmentNumber") is not None for e in here) and k > term_now():
            raise Err("InvalidInputError", "ALG.EVENTS.ANCHOR", k)
        reps = [e for e in here if e["type"] == "ReportedBalance"]
        for e in reps:
            anchors.append({"eventId": e["id"], "k": k, "reported": e["balance"],
                            "projected": txt(B), "realDelta": txt(num(e["balance"]) - B)})
        if reps:
            B = num(sorted(reps, key=lambda e: (e["date"], e["id"]))[-1]["balance"])
        for e in here:
            if e["type"] == "RateChange":
                t_now = term_now()
                if e.get("interestRate") is not None:
                    i = num(e["interestRate"])
                if e.get("insuranceRates") is not None:
                    comps = [num(x) for x in e["insuranceRates"]]
                rate = (i + sum(comps, F(0))) / 12
                if e["policy"] == "RECALC_INSTALLMENT_KEEP_TERM":
                    L, fixed_term = lvl(B, rate, t_now - k + 1), t_now
                else:
                    if e["policy"] == "BANK_INSTALLMENT":
                        L = num(e["bankInstallment"])
                    if L - hu2(B * rate) <= 0:
                        rule = "ALG.RATE.BANK_INSTALLMENT" if e["policy"] == "BANK_INSTALLMENT" \
                            else "ALG.RATE.KEEP_INSTALLMENT"
                        raise Err("NegativeAmortizationError", rule, k)
                    fixed_term = None
            elif e["type"] == "FixedChargeChange":
                charges = [(num(c["amount"]), d) for c in e["fixedCharges"]]
        it, ps, cap, fc = compute_row(B, i, comps, L, prof)
        is_last = L - fc >= B or (fixed_term is not None and k >= fixed_term)
        if is_last:
            it, ps = final_row(B, i, comps, prof)
            cap = B
        elif fixed_term is None and cap <= 0:
            raise Err("NegativeAmortizationError", "ALG.TERM", k)
        fx = sum((a for a, eff in charges if eff <= d), F(0))
        ins = sum(ps, F(0))
        row = {"k": k, "dueDate": d, "opening": B, "level": L, "interest": it, "insurance": ins,
               "insuranceComponents": ps, "capital": cap, "fixedCharges": fx, "total": cap + it + ins + fx,
               "closing": B - cap, "prepayment": F(0), "commission": F(0), "paid": False}
        B = B - cap
        for e in here:
            if e["type"] not in ("Prepayment", "AdvanceInstallments") or B == 0:
                continue
            if e["type"] == "Prepayment":
                want, mode = num(e["amount"]), e["mode"]
            else:
                want, mode = sum(count_rest(B, i, comps, L, prof, k, fixed_term)[:e["count"]], F(0)), "REDUCE_TERM"
            paid_now = want if want < B else B
            c = e.get("commission")
            fee = F(0) if c is None else (num(c["amount"]) if c["kind"] == "FLAT" else hu2(paid_now * num(c["rate"])))
            t_before = fixed_term if fixed_term is not None else k + len(count_rest(B, i, comps, L, prof, k, None))
            B -= paid_now
            row["prepayment"] += paid_now
            row["commission"] += fee
            if B == 0:
                break
            if mode == "REDUCE_TERM":
                fixed_term = None
            else:
                L, fixed_term = lvl(B, (i + sum(comps, F(0))) / 12, t_before - k), t_before
        for e in here:
            if e["type"] == "ActualPayment":
                row["paid"] = True
                bd = e.get("breakdown")
                if bd:
                    proj = {"capital": cap, "interest": it, "insurance": ins, "fixedCharges": fx}
                    entry = {"eventId": e["id"], "k": k}
                    for name in ("capital", "interest", "insurance", "fixedCharges"):
                        entry[name] = txt(num(bd[name]) - proj[name])
                    deltas.append(entry)
        rows.append(row)
        if is_last:
            break
    beyond = [kk for (kk, *_), e in keyed if kk > k and e.get("installmentNumber") is not None]
    if beyond:
        raise Err("InvalidInputError", "ALG.EVENTS.ANCHOR", min(beyond))
    return {"rows": rows, "anchors": anchors, "componentDeltas": deltas}


def row_out(r: dict) -> dict:
    return {key: (val if key in ("k", "dueDate", "paid") else
                  [txt(x) for x in val] if key == "insuranceComponents" else txt(val)) for key, val in r.items()}


def sums(rows: list) -> dict:
    def s(key):
        return sum((r[key] for r in rows), F(0))
    return {"interest": txt(s("interest")), "insurance": txt(s("insurance")), "capital": txt(s("capital")),
            "fixedCharges": txt(s("fixedCharges")), "prepayments": txt(s("prepayment")),
            "commissions": txt(s("commission")), "total": txt(s("total")),
            "totalPaid": txt(s("total") + s("prepayment") + s("commission"))}


def per_year(rows: list) -> list:
    out: dict = {}
    for r in rows:
        y = int(r["dueDate"][:4])
        acc = out.setdefault(y, [F(0)] * 7)
        for j, key in enumerate(("capital", "interest", "insurance", "fixedCharges", "prepayment",
                                 "commission", "total")):
            acc[j] += r[key]
    names = ("capital", "interest", "insurance", "fixedCharges", "prepayments", "commissions", "total")
    return [dict({"year": y}, **{nm: txt(v) for nm, v in zip(names, acc)}) for y, acc in sorted(out.items())]


def metrics(base_rows: list, scen_rows: list, currency: str) -> dict:
    """ComparisonMetrics: [ALG.METRICS] of a scenario against its base."""
    b, s = sums(base_rows), sums(scen_rows)
    paid_b, paid_s = num(b["totalPaid"]), num(s["totalPaid"])
    return {"currency": currency,
            "interestSaved": txt(num(b["interest"]) + num(b["insurance"]) - num(s["interest"]) - num(s["insurance"])),
            "monthsSaved": len(base_rows) - len(scen_rows), "baseEndDate": base_rows[-1]["dueDate"],
            "endDate": scen_rows[-1]["dueDate"], "baseTotalPaid": txt(paid_b), "totalPaid": txt(paid_s),
            "netSaving": txt(paid_b - paid_s)}


def schedule_out(res: dict, shown_ks: list, with_yearly: bool) -> dict:
    rows = res["rows"]
    out = {"installmentCount": len(rows), "endDate": rows[-1]["dueDate"],
           "rows": [row_out(rows[k - 1]) for k in shown_ks], "totals": sums(rows)}
    if with_yearly:
        out["yearly"] = per_year(rows)
    return out


def real_delta(res: dict) -> dict:
    return {"perAnchor": res["anchors"], "perComponent": res["componentDeltas"]}


def shown(exp: dict) -> list:
    return [r["k"] for r in exp.get("rows", [])]


def cut(terms: dict, real: list) -> int:
    ks = [k_of(terms, e) for e in real if e["type"] in
          ("ReportedBalance", "ActualPayment", "Prepayment", "AdvanceInstallments")]
    return max(ks) if ks else 0


def goal_check(case: dict, exp: dict) -> tuple:
    t, req = case["terms"], case["request"]
    real = case["realEvents"]
    k = k_of(t, {"type": "Prepayment", "date": req["prepaymentDate"]})
    if k <= cut(t, real):
        raise Err("InfeasibleGoalError", "ALG.GOAL", k)
    extra = (case["scenarioEvents"] or []) if req["basePath"] == "SCENARIO" else []
    base_events = real + extra
    base = run(t, base_events)["rows"]
    g = req["goal"]
    mode = "REDUCE_TERM" if g["kind"] == "FINISH_BY" else "REDUCE_INSTALLMENT"

    def with_amount(a: F) -> list:
        p = {"id": "goal-seek", "type": "Prepayment", "date": req["prepaymentDate"], "amount": txt(a), "mode": mode}
        return run(t, base_events + [p])["rows"]

    def ok(rows: list) -> bool:
        if g["kind"] == "FINISH_BY":
            return rows[-1]["dueDate"] <= g["date"]
        return len(rows) <= k or rows[k]["total"] <= num(g["amount"])

    closing = base[k - 1]["closing"]
    context = {"k": k, "closingK": txt(closing)}
    if ok(base):
        return {"kind": "ALREADY_MET"}, context
    if not ok(with_amount(closing)):
        return {"kind": "INFEASIBLE", "payoffAmount": txt(closing), "reason": "GOAL_DATE_BEFORE_PREPAYMENT"}, context
    amount = num(exp.get("amount", "0.00"))
    if amount <= 0 or amount > closing or not ok(with_amount(amount)):
        raise AssertionError(f"FOUND amount {exp.get('amount')} does not meet the goal")
    if amount > F(1, 100) and ok(with_amount(amount - F(1, 100))):
        raise AssertionError(f"FOUND amount {exp.get('amount')} is not minimal")
    return {"kind": "FOUND", "amount": txt(amount), "metrics": metrics(base, with_amount(amount), t["currency"]),
            "isPayoff": amount == closing}, context


def validate(case: dict) -> tuple:
    t, real, rep = case["terms"], case["realEvents"], case["reported"]
    k = k_of(t, rep)
    keep = [e for e in real if e["type"] != "ReportedBalance" or k_of(t, e) < k]
    op = {r["k"]: r["opening"] for r in run(t, keep)["rows"]}
    br = num(rep["balance"])
    delta = br - op[k]
    limit = max(F(50), F(2, 10000) * br)
    if abs(delta) <= 1:
        status, cause = "GREEN", None
    else:
        status = "AMBER" if abs(delta) <= limit else "RED"
        cause = "INSTALLMENT_MISALIGNMENT" if any(abs(br - op[j]) <= 1 for j in (k - 1, k + 1) if j in op) \
            else "UNKNOWN"
    return ({"k": k, "reported": txt(br), "modeled": txt(op[k]), "realDelta": txt(delta), "status": status,
             "cause": cause},
            {"modeledPrevious": txt(op[k - 1]) if k - 1 in op else None,
             "modeledNext": txt(op[k + 1]) if k + 1 in op else None, "amberLimit": txt(limit, 6)})


def recompute_case(case: dict) -> tuple:
    """Returns (expected, context) recomputed from the case inputs; context is None when it does not apply."""
    exp, t = case["expected"], case["terms"]
    try:
        if case["operation"] == "buildSchedule":
            res = run(t, case["events"])
            out = schedule_out(res, shown(exp), "yearly" in exp)
            if res["anchors"] or res["componentDeltas"]:
                out["realDelta"] = real_delta(res)
            if "comparedToNoEvents" in exp:
                out["comparedToNoEvents"] = metrics(run(t, [])["rows"], res["rows"], t["currency"])
            return out, None
        if case["operation"] == "buildPaths":
            ck = cut(t, case["realEvents"])
            for e in case["scenarioEvents"]:
                if k_of(t, e) <= ck:
                    raise Err("InvalidInputError", "ALG.PATHS.CUTOFF", k_of(t, e))
            real = run(t, case["realEvents"])
            scen = run(t, case["realEvents"] + case["scenarioEvents"])
            original = run(t, [])["rows"]
            return {"cutoffK": ck,
                    "original": {"installmentCount": len(original), "endDate": original[-1]["dueDate"],
                                 "totals": sums(original)},
                    "real": schedule_out(real, shown(exp["real"]), False),
                    "scenario": schedule_out(scen, shown(exp["scenario"]), "yearly" in exp["scenario"]),
                    "realDelta": real_delta(real),
                    "metrics": metrics(real["rows"], scen["rows"], t["currency"])}, None
        if case["operation"] == "goalSeek":
            return goal_check(case, exp)
        if case["operation"] == "validateAgainstReportedBalance":
            return validate(case)
    except Err as err:
        return {"error": err.as_json}, None
    raise ValueError(case["operation"])


def diff(a, b, path="") -> list:
    if isinstance(a, dict) and isinstance(b, dict):
        out = []
        for key in sorted(set(a) | set(b)):
            out += diff(a.get(key, "<missing>"), b.get(key, "<missing>"), f"{path}.{key}")
        return out
    if isinstance(a, list) and isinstance(b, list) and len(a) == len(b):
        out = []
        for j, (x, y) in enumerate(zip(a, b)):
            out += diff(x, y, f"{path}[{j}]")
        return out
    return [] if a == b and type(a) is type(b) else [f"{path}: file={a!r} recomputed={b!r}"]


def main(root: str) -> int:
    bad = 0
    for path in sorted(Path(root).glob("*/*.json")):
        doc = json.loads(path.read_text(encoding="utf-8"))
        problems = []
        for case in doc["cases"]:
            try:
                expected, context = recompute_case(case)
                problems += [f"{case['id']}{d}" for d in diff(case["expected"], expected)]
                problems += [f"{case['id']}.context{d}" for d in diff(case.get("context"), context)]
            except AssertionError as err:
                problems.append(f"{case['id']}: {err}")
        if problems:
            bad += 1
            print(f"{doc['id']}: recomputed: NO")
            for p in problems[:20]:
                print(f"  {p}")
        else:
            print(f"{doc['id']}: recomputed: yes")
    return 1 if bad else 0


if __name__ == "__main__":
    sys.exit(main(sys.argv[1]))
```

```python
"""Inserts <a id="alg-…"></a> before every [ALG.*] definition in docs/algorithm.md (idempotent).

Usage: python3 add_anchors.py PATH_TO_ALGORITHM_MD
A definition is a heading whose text starts with [ALG.X] or a bold **[ALG.X…**.
"""
from __future__ import annotations

import re
import sys
from pathlib import Path

HEADING_RE = re.compile(r"^(#+ )(\[(ALG\.[A-Z0-9_.]+)\])")
BOLD_RE = re.compile(r"(<a id=\"[a-z0-9-]+\"></a>)?\*\*\[(ALG\.[A-Z0-9_.]+)\]")


def slug(rule_id: str) -> str:
    return rule_id.lower().replace(".", "-").replace("_", "-")


def anchor_line(line: str) -> str:
    m = HEADING_RE.match(line)
    if m:
        return f'{m.group(1)}<a id="{slug(m.group(3))}"></a>{line[len(m.group(1)):]}'

    def bold(mm: re.Match) -> str:
        if mm.group(1):
            return mm.group(0)
        return f'<a id="{slug(mm.group(2))}"></a>**[{mm.group(2)}]'

    return BOLD_RE.sub(bold, line)


def add_anchors(text: str) -> str:
    return "\n".join(anchor_line(ln) for ln in text.split("\n"))


if __name__ == "__main__":
    p = Path(sys.argv[1])
    before = p.read_text(encoding="utf-8")
    after = add_anchors(before)
    p.write_text(after, encoding="utf-8")
    print(f"anchors: {after.count('<a id=')} (added {after.count('<a id=') - before.count('<a id=')})")
```

```python
def walk(node, key, where, errs):
    if isinstance(node, dict):
        for k2, v2 in node.items():
            walk(v2, k2, f"{where}.{k2}", errs)
    elif isinstance(node, list):
        for j, v2 in enumerate(node):
            walk(v2, key, f"{where}[{j}]", errs)
    elif isinstance(node, bool):
        if key not in BOOL_KEYS:
            errs.append(f"{where}: boolean not allowed under '{key}'")
    elif isinstance(node, int):
        if key not in INT_KEYS:
            errs.append(f"{where}: JSON integer not allowed under '{key}'")
    elif isinstance(node, float):
        errs.append(f"{where}: JSON float {node!r}; use a decimal string")
    elif isinstance(node, str):
        if key in MONEY_KEYS and not MONEY_RE.match(node):
            errs.append(f"{where}: money '{node}' must be a decimal string with 2 decimals")
        if key in RATE_KEYS and not DECIMAL_RE.match(node):
            errs.append(f"{where}: rate '{node}' must be a decimal string")


def fields(where: str, obj, required: set, optional: set = frozenset()) -> list:
    """Exact key set of one object: required keys present, nothing outside required | optional."""
    if not isinstance(obj, dict):
        return [f"{where}: object expected"]
    errs = [f"{where}: unknown key '{k}'" for k in sorted(set(obj) - set(required) - set(optional))]
    return errs + [f"{where}: missing key '{k}'" for k in sorted(set(required) - set(obj))]


def by_kind(where: str, obj, table: dict) -> list:
    if not isinstance(obj, dict) or obj.get("kind") not in table:
        return [f"{where}: kind must be one of {', '.join(table)}"]
    return fields(where, obj, table[obj["kind"]])


def check_event(where: str, e, lenient: bool) -> list:
    """One DomainEvent: known type, only its fields, no nulls; required fields unless the case expects an error."""
    kind = e.get("type") if isinstance(e, dict) else None
    if kind not in EVENT_FIELDS:
        return [f"{where}: unknown event type {kind!r}"]
    required, optional = EVENT_FIELDS[e["type"]]
    errs = []
    for key, val in e.items():
        if val is None:
            errs.append(f"{where}.{key} is null; omit optional fields")
        elif key not in EVENT_BASE | required | optional:
            errs.append(f"{where}: unknown field '{key}' for {e['type']}")
    if not lenient:
        errs += [f"{where}: missing field '{key}' for {e['type']}" for key in sorted((EVENT_BASE | required) - set(e))]
    if isinstance(e.get("commission"), dict):
        errs += by_kind(f"{where}.commission", e["commission"], BY_KIND["commission"])
    if isinstance(e.get("breakdown"), dict):
        errs += fields(f"{where}.breakdown", e["breakdown"], NESTED["perComponent"] - {"eventId", "k"})
    if e["type"] == "FixedChargeChange":
        for j, c in enumerate(e.get("fixedCharges") or []):
            errs += fields(f"{where}.fixedCharges[{j}]", c, {"label", "amount"})
    return errs


def nested(where: str, node) -> list:
    """Every named output object (rows, totals, metrics, realDelta lists, yearly, original) has its exact keys."""
    errs = []
    if isinstance(node, list):
        for j, item in enumerate(node):
            errs += nested(f"{where}[{j}]", item)
    elif isinstance(node, dict):
        for key, val in node.items():
            if key == "realDelta" and isinstance(val, dict):

                    errs += fields(f"{where}.{key}" + (f"[{j}]" if isinstance(val, list) else ""), item, NESTED[key])
            errs += nested(f"{where}.{key}", val)
    return errs

def event_types(doc: dict) -> list:
    types = set()
    for case in doc.get("cases", []):
        for lst in EVENT_LISTS:
            types.update(e["type"] for e in (case.get(lst) or []))
        if case.get("operation") in IMPLICIT:
            types.add(IMPLICIT[case["operation"]])
    return sorted(types)


def check_json(path: Path, set_name: str, ids: set) -> list:
    rel = f"{set_name}/{path.name}"
    try:
        doc = json.loads(path.read_text(encoding="utf-8"))
    except ValueError as err:
        return [f"{rel}: invalid JSON ({err})"]
    errs = []
    if doc.get("synthetic") is not True:
        errs.append(f"{rel}: root synthetic must be true")
    if doc.get("schema") != SCHEMA:
        errs.append(f"{rel}: schema must be '{SCHEMA}'")
    if doc.get("id") != path.stem:
        errs.append(f"{rel}: id must equal the file name")
    for sec in doc.get("sections", []):
        if sec not in ids:
            errs.append(f"{rel}: section {sec} is not defined in algorithm.md")
    expected_types = event_types(doc)
    if doc.get("eventTypes") != expected_types:
        errs.append(f"{rel}: eventTypes {doc.get('eventTypes')} != {expected_types}")
    if set_name == "core" and expected_types:
        errs.append(f"{rel}: core/ files must be event-free")
    if set_name == "events" and not expected_types:
        errs.append(f"{rel}: events/ files must use at least one event")
    if not doc.get("cases"):
        errs.append(f"{rel}: no cases")
    for case in doc.get("cases", []):
        errs += check_case(rel, case)
    walk({k: v for k, v in doc.items() if k != "pendingItems"}, "", rel, errs)
    if not all(isinstance(x, int) and not isinstance(x, bool) for x in doc.get("pendingItems", [None])):
        errs.append(f"{rel}: pendingItems must be a list of integers")
    return errs


def table(md: str, heading: str) -> list:
    m = re.search(rf"(?ms)^## {re.escape(heading)}.*?(?=^## |\Z)", md)
    if not m:
        return []
    rows = [ln for ln in m.group(0).splitlines() if ln.startswith("|")][2:]
    return [[c.strip() for c in ln.strip().strip("|").split("|")] for ln in rows]


def check_index(root: Path, alg: str, docs: dict) -> list:
    index_path = root / EX_DIR / "INDEX.md"
    if not index_path.exists():
        return ["INDEX.md is missing"]
    md = index_path.read_text(encoding="utf-8")
    errs = []
    anchors = set(ANCHOR_RE.findall(alg))
    for target in LINK_RE.findall(md):
        file_part, _, frag = target.partition("#")
        if target.startswith("http"):
            continue
        dest = (index_path.parent / file_part).resolve()
        if not dest.exists():
            errs.append(f"INDEX.md: broken link {target}")
        elif frag and dest.name == "algorithm.md" and frag not in anchors:
            errs.append(f"INDEX.md: anchor #{frag} does not exist in algorithm.md")
    checklist = table(md, "Checklist")
    by_item: dict = {}
    for cells in checklist:
        files = [t for t in LINK_RE.findall(cells[3]) if t.endswith(".json")] if len(cells) > 3 else []
        rules = [t for t in LINK_RE.findall(cells[2]) if "algorithm.md#" in t] if len(cells) > 2 else []
        if not files or not rules:
            errs.append(f"INDEX.md checklist row '{cells[0]}' needs rule anchors and example files")
        if len(cells) < 5 or cells[4] != "sí":
            errs.append(f"INDEX.md checklist row '{cells[0]}' is not marked 'sí' in Recalculado")
        by_item[cells[0]] = files
    for item in ["EX"] + [str(n) for n in pending_items(alg)]:
        files = by_item.get(item)
        if not files:
            errs.append(f"INDEX.md: [ALG.PENDING] item {item} is unmapped")
            continue
        for f in files:
            doc = docs.get(f)
            if doc is None:
                errs.append(f"INDEX.md: item {item} links {f}, which is not an example file")
            elif item != "EX" and int(item) not in doc.get("pendingItems", []):
                errs.append(f"INDEX.md: {f} does not list item {item} in pendingItems")
            elif item == "EX" and "ALG.EXAMPLE" not in doc.get("sections", []):
                errs.append(f"INDEX.md: {f} does not cite ALG.EXAMPLE")
    listed = {}
    for cells in table(md, "Archivos"):
        links = LINK_RE.findall(cells[0])
        if links:
            listed[links[0]] = cells[2]
    for f, doc in docs.items():
        if f not in listed:
            errs.append(f"INDEX.md: {f} is not listed under '## Archivos'")
            continue
        shown = "—" if not doc.get("eventTypes") else ", ".join(doc["eventTypes"])
        if listed[f] != shown:
            errs.append(f"INDEX.md: events of {f} are '{listed[f]}', expected '{shown}'")
    return errs


PARTS = ("algorithm", "glossary", "examples", "index", "all")


def run_checks(root: Path, part: str = "all") -> list:
    alg = (root / "docs/algorithm.md").read_text(encoding="utf-8")
    ids = set(defined_ids(alg))
    errs = []
    if part in ("algorithm", "all"):
        errs += check_algorithm(alg)
    if part in ("glossary", "all"):
        errs += check_glossary(alg, (root / "docs/glossary.md").read_text(encoding="utf-8"))
    if part in ("examples", "index", "all"):
        docs = {}
        for set_name in ("core", "events"):
            for path in sorted((root / EX_DIR / set_name).glob("*.json")):
                if part != "index":
                    errs += check_json(path, set_name, ids)
                docs[f"{set_name}/{path.name}"] = json.loads(path.read_text(encoding="utf-8"))
        if part != "index":
            if not docs:
                errs.append(f"no example files under {EX_DIR}/core and {EX_DIR}/events")
            others = [p for p in (root / EX_DIR).rglob("*") if p.is_file() and p.name != "INDEX.md"
                      and p.parent.name not in ("core", "events")]
            errs += [f"unexpected file {p.relative_to(root)}" for p in others]
        if part in ("index", "all"):
            errs += check_index(root, alg, docs)
    return errs


def main(argv: list) -> int:
    root, part = Path(argv[1]), (argv[2] if len(argv) > 2 else "all")
    if part not in PARTS:
        print(f"ERROR: part must be one of {', '.join(PARTS)}")
        return 2
    errs = run_checks(root, part)
    for e in errs:
        print(f"ERROR: {e}")
    if not errs:
        print(f"OK: {part}")
    return 1 if errs else 0


if __name__ == "__main__":
    sys.exit(main(sys.argv))
```

```python
    alg = (root / "docs/algorithm.md").read_text(encoding="utf-8")
    log = Path(log_path).read_text(encoding="utf-8")
    ok = set(re.findall(r"(?m)^(\S+): recomputed: yes$", log))
    docs = {}
    for set_name in ("core", "events"):
        for p in sorted((root / EX_DIR / set_name).glob("*.json")):
            docs[f"{set_name}/{p.name}"] = json.loads(p.read_text(encoding="utf-8"))
    items = [("EX", "Ejemplo base de [ALG.EXAMPLE]: calendario sin eventos y sus abonos de ejemplo")]
    items += [(str(n), t) for n, t in sorted(pending_texts(alg).items())]
    out = [HEADER, "## Checklist de [ALG.PENDING]", "",
           "| Ítem | Qué cubre | Reglas | Ejemplos | Recalculado |", "|---|---|---|---|---|"]
    for item, text in items:
        files = [f for f, d in docs.items()
                 if (item == "EX" and d["id"].startswith("ex00-")) or (item != "EX" and int(item) in d["pendingItems"])]
        secs = []
        for f in files:
            secs += [s for s in docs[f]["sections"] if s not in secs]
        recomputed = "sí" if files and all(docs[f]["id"] in ok for f in files) else "no"
        out.append(f"| {item} | {text} | {', '.join(rule_link(s) for s in secs)} | "
                   f"{', '.join(f'[{f}]({f})' for f in files)} | {recomputed} |")
    out += ["", "## Archivos", "", "| Archivo | Conjunto | Eventos | Casos |", "|---|---|---|---|"]
    for f, d in docs.items():
        events = ", ".join(d["eventTypes"]) if d["eventTypes"] else "—"
        cases = ", ".join(f"`{c['id']}`" for c in d["cases"])
        out.append(f"| [{f}]({f}) | {f.split('/')[0]} | {events} | {cases} |")
    out += ["", FORMAT, RULES]
    (root / EX_DIR / "INDEX.md").write_text("\n".join(out).rstrip() + "\n", encoding="utf-8")
    print(f"INDEX.md: {len(items)} checklist rows, {len(docs)} files")


if __name__ == "__main__":
```

### Decisiones que fija esta tarjeta y preguntas para Opus

Las fija esta tarjeta porque los ejemplos no se podían calcular sin ellas; ambos motores las aplican igual y el revisor debe confirmarlas:

1. **`MAX_INSTALLMENT` nunca es `INFEASIBLE`.** `[ALG.GOAL]` cuenta la liquidación en `k` como cumplida, así que el ítem 11 de `[ALG.PENDING]` («`INFEASIBLE` para ambas metas») contradecía la regla; se corrige el ítem (Task W0-02.5, Step 7) y el caso `max-installment-found-payoff` muestra `FOUND` con `isPayoff`.
2. **Plazo fijo con `level = 0.00`:** no es error; la cuota `term` liquida. Sin esta regla, la búsqueda `MAX_INSTALLMENT` cerca de la liquidación lanzaba `NegativeAmortizationError` (se vio al calcular).
3. **Plazo derivado con `level − financialCharge ≤ 0`** en una cuota (posible tras un ancla que sube el saldo): `NegativeAmortizationError` con regla `ALG.TERM`, porque el plazo no terminaría.
4. **`term` vigente en la fase 1 con plazo derivado:** se calcula con el saldo ya re-anclado por la fase 0.
5. **Errores con regla:** los ejemplos nombran la clase y el id `ALG.*` que la exige, en lugar de códigos nuevos.
6. **Columnas anuales:** se agregan `commissions`; `total` no incluye abonos ni comisiones.
7. **Nombres de los contratos congelados.** Las entradas usan los tipos de entrada de W0-03 (`LoanTerms`, `DomainEvent`, `PathsInput`, `GoalSeekRequest`, `TemplateValidationRequest`) con los campos de las cargas de `LoanEvent` de W0-04: `insuranceComponents`, `bankInstallment`, `count`, `total` en `ActualPayment`, `commission` omitida cuando no hay, `scenarioEvents`, `goal = {kind, date | amount}`. `expected` usa los nombres de las salidas del dominio (`installmentCount`, `ScheduleTotals` sin totales por componente, `RealDelta`, `ComparisonMetrics`, `GoalSeekResult` con `kind`, `TemplateValidationResult`), y lo que el dominio no devuelve va en `context`. 8. **Desglose completo de `ActualPayment`.** `ActualPaymentBreakdown` (W0-03) y `paymentBreakdownSchema` (W0-04) exigen los cuatro componentes, así que `[ALG.ACTUAL]` ya no admite desgloses parciales: `ex09` trae dos desgloses completos y `perComponent` lleva siempre los cuatro.
